    static class Player {
        String name;
        String playerClass;
        int level = 1;
        int exp = 0;
        int expToNextLevel = 100;
        int health = 100;
        int maxHealth = 100;
        int mana = 50;
        int maxMana = 50;
        int attack = 10;
        int defense = 5;
        int gold = 50;
        int potions = 3;
        ArrayList<String> inventory = new ArrayList<>();

        public Player(String name, String playerClass) {
            this.name = name;
            this.playerClass = playerClass;
            setClassStats();
        }

        private void setClassStats() {
            switch(playerClass) {
                case "Воин":
                    maxHealth = 120;
                    health = 120;
                    attack = 15;
                    defense = 10;
                    break;
                case "Маг":
                    maxHealth = 80;
                    health = 80;
                    attack = 8;
                    defense = 4;
                    maxMana = 100;
                    mana = 100;
                    break;
                case "Лучник":
                    maxHealth = 100;
                    health = 100;
                    attack = 12;
                    defense = 7;
                    break;
            }
        }

        public void addExp(int amount) {
            exp += amount;
            if (exp >= expToNextLevel) {
                levelUp();
            }
        }

        private void levelUp() {
            level++;
            exp = 0;
            expToNextLevel = (int)(expToNextLevel * 1.5);

            maxHealth += 20;
            maxMana += 10;
            attack += 3;
            defense += 2;
            health = maxHealth;
            mana = maxMana;

            potions += 2;
        }

        public void heal(int amount) {
            health = Math.min(maxHealth, health + amount);
        }

        public void takeDamage(int damage) {
            int actualDamage = Math.max(1, damage - defense / 2);
            health -= actualDamage;
        }

        public boolean isAlive() {
            return health > 0;
        }
    }

        String name;
        int health;
        int maxHealth;
        int attack;
        int defense;
        int expReward;

        String imagePath;

        public Enemy(String name, int health, int attack, int defense, int expReward, int goldReward) {
            this.name = name;
            this.maxHealth = health;
            this.health = health;
            this.attack = attack;
            this.defense = defense;
            this.expReward = expReward;
            this.goldReward = goldReward;
        }

        public void takeDamage(int damage) {
            int actualDamage = Math.max(1, damage - defense / 4);
            health -= actualDamage;
        }

        public boolean isAlive() {
            return health > 0;
        }
    }

    private void startNewGame() {
        String name = nameField.getText().trim();
        if (name.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Введите имя героя!", "Ошибка", JOptionPane.ERROR_MESSAGE);
            return;
        }

        String playerClass = (String) classComboBox.getSelectedItem();
        player = new Player(name, playerClass);

        // Обновляем интерфейс
        playerNameLabel.setText("Имя: " + player.name);
        playerClassLabel.setText("Класс: " + player.playerClass);
        updateStatsDisplay();

        // Переходим к игре
        cardLayout.show(mainPanel, "game");
        log("Добро пожаловать, " + player.name + "!");
        log("Вы - " + player.playerClass + ", готовый к приключениям!");
    }

    private void exploreLocation(String locationName) {
        log("Вы отправились в " + locationName + "...");

        // Создаем врага в зависимости от локации
        Enemy enemy = null;
        switch(locationName) {
            case "Тёмный лес":
                enemy = new Enemy("Гоблин", 40, 8, 3, 25, 10);
                enemyImageLabel.setText("👹");
                break;
            case "Пещера":
                enemy = new Enemy("Орк", 60, 12, 5, 40, 20);
                enemyImageLabel.setText("👺");
                break;
            case "Подземелье":
                enemy = new Enemy("Дракон", 150, 25, 15, 100, 75);
                enemyImageLabel.setText("🐉");
                break;
        }

        if (enemy != null) {
            enemyNameLabel.setText(enemy.name);
            enemyHealthBar.setMaximum(enemy.maxHealth);
            enemyHealthBar.setValue(enemy.health);
            enemyHealthBar.setString(enemy.health + "/" + enemy.maxHealth);

            // Переходим к битве
            cardLayout.show(mainPanel, "battle");
            startBattle(enemy);
        }
    }

    private void startBattle(Enemy enemy) {
        battleLog.setText("");
        battleLog("Начало боя с " + enemy.name + "!");

        // Сбрасываем обработчики
        for (ActionListener al : attackButton.getActionListeners()) {
            attackButton.removeActionListener(al);
        }
        for (ActionListener al : defendButton.getActionListeners()) {
            defendButton.removeActionListener(al);
        }
        for (ActionListener al : potionButton.getActionListeners()) {
            potionButton.removeActionListener(al);
        }
        for (ActionListener al : runButton.getActionListeners()) {
            runButton.removeActionListener(al);
        }

        // Новые обработчики
        attackButton.addActionListener(e -> performAttack(enemy));
        defendButton.addActionListener(e -> performDefend(enemy));
        potionButton.addActionListener(e -> usePotion(enemy));
        runButton.addActionListener(e -> attemptRun(enemy));
    }

    private void performAttack(Enemy enemy) {
        if (!player.isAlive() || !enemy.isAlive()) return;

        int damage = player.attack + random.nextInt(6);
        if (random.nextInt(100) < 20) {
            damage *= 2;
            battleLog("🔥 КРИТИЧЕСКИЙ УДАР!");
        }

        enemy.takeDamage(damage);
        battleLog("Вы нанесли " + damage + " урона " + enemy.name + "!");

        enemyHealthBar.setValue(enemy.health);
        enemyHealthBar.setString(enemy.health + "/" + enemy.maxHealth);

        if (!enemy.isAlive()) {
            endBattleVictory(enemy);
            return;
        }

        enemyTurn(enemy);
    }

    private void performDefend(Enemy enemy) {
        if (!player.isAlive() || !enemy.isAlive()) return;

        int tempDefense = player.defense + 5;
        battleLog("🛡 Вы защищаетесь! Защита увеличена на 5.");

        enemyTurn(enemy, tempDefense);
    }

    private void usePotion(Enemy enemy) {
        if (player.potions > 0) {
            player.heal(30);
            player.potions--;
            battleLog("🧪 Использовано зелье здоровья! +30 HP");
            updateStatsDisplay();
            enemyTurn(enemy);
        } else {
            battleLog("❌ У вас нет зелий!");
        }
    }

    private void attemptRun(Enemy enemy) {
        if (random.nextInt(100) < 50) {
            battleLog("🏃‍♂️ Вы успешно сбежали!");
            cardLayout.show(mainPanel, "game");
            log("Вы сбежали от " + enemy.name);
        } else {
            battleLog("❌ Не удалось сбежать!");
            enemyTurn(enemy);
        }
    }

    private void enemyTurn(Enemy enemy) {
        enemyTurn(enemy, player.defense);
    }

    private void enemyTurn(Enemy enemy, int currentDefense) {
        if (!player.isAlive() || !enemy.isAlive()) return;

        int enemyDamage = Math.max(1, enemy.attack - currentDefense / 2 + random.nextInt(4));
        player.takeDamage(enemyDamage);
        battleLog(enemy.name + " наносит вам " + enemyDamage + " урона!");

        updateStatsDisplay();

        if (!player.isAlive()) {
            endBattleDefeat();
        }
    }

    private void endBattleVictory(Enemy enemy) {
        battleLog("\n🎊 ПОБЕДА! " + enemy.name.toUpperCase() + " ПОБЕЖДЕН!");
        battleLog("Получено " + enemy.expReward + " опыта и " + enemy.goldReward + " золота!");

        player.addExp(enemy.expReward);
        player.gold += enemy.goldReward;

        // Шанс найти зелье
        if (random.nextInt(100) < 30) {
            player.potions++;
            battleLog("🧪 Найдено зелье здоровья!");
        }

        updateStatsDisplay();

        // Кнопка продолжения
        javax.swing.Timer timer = new javax.swing.Timer(2000, e -> {
            cardLayout.show(mainPanel, "game");
            log("Вы победили " + enemy.name + "!");
            log("Получено: " + enemy.expReward + " опыта, " + enemy.goldReward + " золота");
        });
        timer.setRepeats(false);
        timer.start();
    }

    private void endBattleDefeat() {
        battleLog("\n💀 ВЫ ПОГИБЛИ...");

        javax.swing.Timer timer = new javax.swing.Timer(2000, e -> {
            int response = JOptionPane.showConfirmDialog(
                    this,
                    "Вы погибли. Начать новую игру?",
                    "Поражение",
                    JOptionPane.YES_NO_OPTION
            );

            if (response == JOptionPane.YES_OPTION) {
                cardLayout.show(mainPanel, "menu");
            } else {
                System.exit(0);
            }
        });
        timer.setRepeats(false);
        timer.start();
    }

    private void visitShop() {
        String[] options = {
                "🧪 Зелье здоровья (15 золота)",
                "⚔ Улучшение меча (50 золота)",
                "🛡 Улучшение брони (50 золота)",
                "🚪 Уйти"
        };

        int choice = JOptionPane.showOptionDialog(
                this,
                "Добро пожаловать в магазин!\nВаше золото: " + player.gold,
                "Магазин",
                JOptionPane.DEFAULT_OPTION,
                JOptionPane.INFORMATION_MESSAGE,
                null,
                options,
                options[3]
        );

        switch(choice) {
            case 0:
                if (player.gold >= 15) {
                    player.gold -= 15;
                    player.potions++;
                    log("Куплено зелье здоровья за 15 золота");
                    updateStatsDisplay();
                } else {
                    JOptionPane.showMessageDialog(this, "Недостаточно золота!");
                }
                break;
            case 1:
                if (player.gold >= 50) {
                    player.gold -= 50;
                    player.attack += 2;
                    log("Куплено улучшение меча! Атака +2");
                    updateStatsDisplay();
                } else {
                    JOptionPane.showMessageDialog(this, "Недостаточно золота!");
                }
                break;
            case 2:
                if (player.gold >= 50) {
                    player.gold -= 50;
                    player.defense += 2;
                    log("Куплено улучшение брони! Защита +2");
                    updateStatsDisplay();
                } else {
                    JOptionPane.showMessageDialog(this, "Недостаточно золота!");
                }
                break;
        }
    }

    private void restAtInn() {
        if (player.gold >= 10) {
            player.gold -= 10;
            player.health = player.maxHealth;
            player.mana = player.maxMana;
            log("Вы отдохнули в таверне за 10 золота");
            log("Здоровье и мана полностью восстановлены!");
            updateStatsDisplay();
        } else {
            JOptionPane.showMessageDialog(this, "Недостаточно золота! Нужно 10 золота.");
        }
    }

    private void showStats() {
        StringBuilder stats = new StringBuilder();
        stats.append("=== ХАРАКТЕРИСТИКИ ===\n");
        stats.append("Имя: ").append(player.name).append("\n");
        stats.append("Класс: ").append(player.playerClass).append("\n");
        stats.append("Уровень: ").append(player.level).append("\n");
        stats.append("Опыт: ").append(player.exp).append("/").append(player.expToNextLevel).append("\n");
        stats.append("Здоровье: ").append(player.health).append("/").append(player.maxHealth).append("\n");
        stats.append("Мана: ").append(player.mana).append("/").append(player.maxMana).append("\n");
        stats.append("Атака: ").append(player.attack).append("\n");
        stats.append("Защита: ").append(player.defense).append("\n");
        stats.append("Золото: ").append(player.gold).append("\n");
        stats.append("Зелья: ").append(player.potions).append("\n");

        if (!player.inventory.isEmpty()) {
            stats.append("\n=== ИНВЕНТАРЬ ===\n");
            for (String item : player.inventory) {
                stats.append("- ").append(item).append("\n");
            }
        }

        JOptionPane.showMessageDialog(this, stats.toString(), "Статистика", JOptionPane.INFORMATION_MESSAGE);
    }
